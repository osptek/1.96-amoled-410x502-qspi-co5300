//此函数用于设置“窗口”---GRAM

void DM_block_write(unsigned int Xstart,unsigned int Xend,unsigned int Ystart,unsigned int Yend)
{	
	
SPI_CS=0;
SPI_WriteComm(0x2a);   
SPI_WriteData(Xstart+22>>8);
SPI_WriteData(Xstart+22&0xff);
SPI_WriteData(Xend+22>>8);
SPI_WriteData(Xend+22&0xff);
SPI_CS=1;
	
SPI_CS=0;
SPI_WriteComm(0x2b);   
SPI_WriteData(Ystart>>8);
SPI_WriteData(Ystart&0xff);
SPI_WriteData(Yend>>8);
SPI_WriteData(Yend&0xff);
SPI_CS=1;

SPI_CS=0;	
SPI_WriteComm(0x2c);
SPI_CS=1;	
	
}
